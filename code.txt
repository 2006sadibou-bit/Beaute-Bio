"use client";

import {
  useEffect,
  useState,
  type FormEvent,
  type ChangeEvent,
} from "react";

type Product = {
  id: string;
  name: string;
  slug: string;
  description: string | null;
  price: number;
  stock: number;
  image_url: string | null;
  category_id: string | null;
  active: boolean;
};

type Category = {
  id: string;
  name: string;
};

const emptyForm = {
  name: "",
  slug: "",
  description: "",
  price: "",
  stock: "0",
  image_url: "",
  category_id: "",
  active: true,
};

export default function AdminProduitsPage() {
  const [connected, setConnected] = useState(false);
  const [pin, setPin] = useState("");
  const [products, setProducts] = useState<Product[]>([]);
  const [categories, setCategories] = useState<Category[]>([]);
  const [editingId, setEditingId] = useState<string | null>(null);
  const [message, setMessage] = useState("");
  const [loading, setLoading] = useState(false);
  const [uploadingImage, setUploadingImage] = useState(false);
  const [imagePreview, setImagePreview] = useState("");
  const [form, setForm] = useState(emptyForm);

  // =====================================================
  // CHARGEMENT DES PRODUITS
  // =====================================================

  async function loadProducts() {
    try {
      const response = await fetch("/api/admin/products");
      const data = await response.json();

      if (!response.ok) {
        setConnected(false);
        setMessage(
          data.details || data.error || "Session expirée."
        );
        return;
      }

      setProducts(data.products || []);
      setCategories(data.categories || []);
      setConnected(true);
    } catch {
      setMessage("Erreur de connexion au serveur.");
    }
  }

  useEffect(() => {
    loadProducts();
  }, []);

  // =====================================================
  // CONNEXION ADMIN
  // =====================================================

  async function login(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    setLoading(true);
    setMessage("");

    try {
      const response = await fetch("/api/admin/login", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ pin }),
      });

      const data = await response.json();

      if (!response.ok) {
        setMessage(
          data.details || data.error || "Code incorrect."
        );
        return;
      }

      setPin("");
      await loadProducts();
    } catch {
      setMessage("Erreur de connexion.");
    } finally {
      setLoading(false);
    }
  }

  // =====================================================
  // APERÇU LOCAL DE LA PHOTO
  // =====================================================

  function readImagePreview(file: File): Promise<string> {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();

      reader.onload = () => {
        if (typeof reader.result === "string") {
          resolve(reader.result);
        } else {
          reject(new Error("Lecture de la photo impossible."));
        }
      };

      reader.onerror = () => {
        reject(new Error("Impossible de lire cette image."));
      };

      reader.readAsDataURL(file);
    });
  }

  // =====================================================
  // ENVOI DE LA PHOTO VERS SUPABASE
  // =====================================================

  async function uploadImage(
    event: ChangeEvent<HTMLInputElement>
  ) {
    const input = event.currentTarget;
    const file = input.files?.[0];

    if (!file) return;

    if (!file.type.startsWith("image/")) {
      setMessage("Choisis un fichier image valide.");
      input.value = "";
      return;
    }

    if (file.size > 5 * 1024 * 1024) {
      setMessage("La photo ne doit pas dépasser 5 Mo.");
      input.value = "";
      return;
    }

    setUploadingImage(true);
    setMessage("Préparation de la photo...");

    try {
      // 1. Afficher immédiatement une vraie prévisualisation.
      const preview = await readImagePreview(file);

      setImagePreview(preview);

      // 2. Envoyer le fichier à l'API.
      setMessage("Envoi de la photo vers Supabase...");

      const formData = new FormData();
      formData.append("file", file);

      const response = await fetch(
        "/api/admin/products/upload",
        {
          method: "POST",
          body: formData,
        }
      );

      const data = await response.json();

      if (!response.ok) {
        setMessage(
          data.error || "Impossible d'envoyer la photo."
        );

        // On conserve l'aperçu local même en cas d'erreur.
        return;
      }

      if (!data.image_url) {
        setMessage(
          "La photo a été envoyée, mais son adresse est introuvable."
        );
        return;
      }

      // 3. Enregistrer l'adresse publique de la photo.
      setForm((previous) => ({
        ...previous,
        image_url: data.image_url,
      }));

      // 4. Remplacer l'aperçu local par l'image Supabase.
      setImagePreview(data.image_url);

      setMessage("Photo envoyée avec succès !");
    } catch (error) {
      console.error("Erreur photo :", error);

      setMessage(
        "Erreur pendant la lecture ou l'envoi de la photo."
      );
    } finally {
      setUploadingImage(false);
      input.value = "";
    }
  }

  // =====================================================
  // ENREGISTREMENT DU PRODUIT
  // =====================================================

  async function saveProduct(
    event: FormEvent<HTMLFormElement>
  ) {
    event.preventDefault();

    if (uploadingImage) {
      setMessage("Attends la fin de l'envoi de la photo.");
      return;
    }

    if (imagePreview && !form.image_url) {
      setMessage(
        "La photo est encore sur ton ordinateur. Réessaie son envoi avant d'enregistrer."
      );
      return;
    }

    setLoading(true);
    setMessage("");

    const payload = {
      ...form,
      price: Number(form.price),
      stock: Number(form.stock),
      category_id: form.category_id || null,
    };

    try {
      const response = await fetch(
        editingId
          ? `/api/admin/products/${editingId}`
          : "/api/admin/products",
        {
          method: editingId ? "PATCH" : "POST",
          headers: {
            "Content-Type": "application/json",
          },
          body: JSON.stringify(payload),
        }
      );

      const data = await response.json();

      if (!response.ok) {
        setMessage(
          data.error || "Enregistrement impossible."
        );
        return;
      }

      setMessage("Produit enregistré avec succès !");

      setForm(emptyForm);
      setImagePreview("");
      setEditingId(null);

      await loadProducts();
    } catch {
      setMessage("Erreur lors de l'enregistrement.");
    } finally {
      setLoading(false);
    }
  }

  // =====================================================
  // MODIFICATION D'UN PRODUIT
  // =====================================================

  function editProduct(product: Product) {
    setEditingId(product.id);

    setForm({
      name: product.name,
      slug: product.slug,
      description: product.description || "",
      price: String(product.price),
      stock: String(product.stock),
      image_url: product.image_url || "",
      category_id: product.category_id || "",
      active: product.active,
    });

    setImagePreview(product.image_url || "");
    setMessage("");

    window.scrollTo({
      top: 0,
      behavior: "smooth",
    });
  }

  // =====================================================
  // ANNULER LA MODIFICATION
  // =====================================================

  function cancelEditing() {
    setEditingId(null);
    setForm(emptyForm);
    setImagePreview("");
    setMessage("");
  }

  // =====================================================
  // SUPPRIMER UN PRODUIT
  // =====================================================

  async function deleteProduct(id: string) {
    if (!confirm("Voulez-vous vraiment supprimer ce produit ?")) {
      return;
    }

    try {
      const response = await fetch(
        `/api/admin/products/${id}`,
        {
          method: "DELETE",
        }
      );

      const data = await response.json();

      if (!response.ok) {
        setMessage(
          data.error || "Suppression impossible."
        );
        return;
      }

      setMessage("Produit supprimé avec succès.");

      await loadProducts();
    } catch {
      setMessage("Erreur pendant la suppression.");
    }
  }

  // =====================================================
  // PUBLIER OU MASQUER UN PRODUIT
  // =====================================================

  async function toggleProduct(product: Product) {
    try {
      const response = await fetch(
        `/api/admin/products/${product.id}`,
        {
          method: "PATCH",
          headers: {
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            active: !product.active,
          }),
        }
      );

      const data = await response.json();

      if (!response.ok) {
        setMessage(
          data.error || "Modification impossible."
        );
        return;
      }

      setMessage(
        product.active
          ? "Produit masqué."
          : "Produit publié."
      );

      await loadProducts();
    } catch {
      setMessage("Erreur pendant la modification.");
    }
  }

  // =====================================================
  // PAGE DE CONNEXION
  // =====================================================

  if (!connected) {
    return (
      <main className="relative flex min-h-screen items-center justify-center overflow-hidden bg-gradient-to-br from-pink-100 via-white to-rose-100 p-5">

        <div className="absolute -left-20 -top-20 h-72 w-72 animate-pulse rounded-full bg-pink-200/50 blur-3xl" />

        <div className="absolute -bottom-20 -right-20 h-72 w-72 animate-pulse rounded-full bg-rose-200/50 blur-3xl" />

        <form
          onSubmit={login}
          className="relative z-10 w-full max-w-md rounded-3xl border border-white bg-white/90 p-8 shadow-2xl backdrop-blur-xl transition-all duration-500 hover:-translate-y-1 hover:shadow-pink-200"
        >
          <div className="mb-6 text-center">

            <div className="mx-auto mb-5 flex h-24 w-24 items-center justify-center rounded-full bg-gradient-to-br from-[#880E4F] to-[#E91E63] text-4xl font-bold text-white shadow-xl transition-transform duration-500 hover:rotate-6 hover:scale-110">
              BG
            </div>

            <h1 className="text-3xl font-extrabold text-[#880E4F]">
              Espace privé Gnilane
            </h1>

            <p className="mt-3 text-gray-500">
              Administration de Beauté Bio
            </p>

            <div className="mx-auto mt-4 h-1 w-20 rounded-full bg-gradient-to-r from-[#880E4F] to-pink-300" />
          </div>

          <label className="mb-2 block text-sm font-semibold text-gray-700">
            Code administrateur
          </label>

          <input
            type="password"
            inputMode="numeric"
            maxLength={6}
            value={pin}
            onChange={(e) =>
              setPin(
                e.target.value.replace(/\D/g, "").slice(0, 6)
              )
            }
            placeholder="Code PIN à 6 chiffres"
            className="w-full rounded-xl border border-pink-200 bg-pink-50/50 p-4 text-center text-xl tracking-[0.5em] outline-none transition-all duration-300 focus:border-[#880E4F] focus:ring-4 focus:ring-pink-100"
            required
          />

          {message && (
            <p className="mt-4 animate-pulse text-center text-sm text-red-600">
              {message}
            </p>
          )}

          <button
            type="submit"
            disabled={loading || pin.length !== 6}
            className="mt-6 w-full rounded-xl bg-gradient-to-r from-[#880E4F] to-[#C2185B] p-4 font-bold text-white shadow-lg transition-all duration-300 hover:-translate-y-1 hover:shadow-xl disabled:cursor-not-allowed disabled:opacity-50"
          >
            {loading ? "Connexion..." : "Se connecter"}
          </button>
        </form>
      </main>
    );
  }

  // =====================================================
  // TABLEAU DE BORD
  // =====================================================

  return (
    <main className="min-h-screen bg-gradient-to-br from-[#FFF5F9] via-white to-pink-50 p-4 md:p-10">

      <div className="mx-auto max-w-7xl">

        {/* EN-TÊTE */}

        <header className="mb-10 flex flex-wrap items-center justify-between gap-5 rounded-3xl bg-white p-6 shadow-lg transition-all duration-300 hover:shadow-xl">

          <div>
            <p className="mb-2 text-sm font-bold tracking-[0.2em] text-[#C2185B]">
              BEAUTÉ BIO BY GNILANE
            </p>

            <h1 className="text-3xl font-extrabold text-[#880E4F] md:text-4xl">
              Administration
            </h1>

            <p className="mt-2 text-gray-500">
              Gérez votre catalogue de produits.
            </p>
          </div>

          <a
            href="/produits"
            className="rounded-xl bg-gradient-to-r from-[#880E4F] to-[#C2185B] px-6 py-3 font-bold text-white shadow-md transition-all duration-300 hover:-translate-y-1 hover:shadow-xl"
          >
            Voir la boutique →
          </a>

        </header>

        {/* MESSAGE */}

        {message && (
          <div className="mb-6 animate-pulse rounded-2xl border border-pink-100 bg-white p-4 text-sm font-semibold text-[#880E4F] shadow-sm">
            {message}
          </div>
        )}

        {/* FORMULAIRE */}

        <section className="mb-10 rounded-3xl border border-pink-100 bg-white p-5 shadow-xl transition-all duration-500 hover:shadow-2xl md:p-8">

          <div className="mb-7 flex flex-wrap items-center justify-between gap-3">

            <div>
              <p className="text-sm font-semibold uppercase tracking-wider text-[#C2185B]">
                Gestion du catalogue
              </p>

              <h2 className="mt-2 text-2xl font-extrabold text-[#880E4F]">
                {editingId
                  ? "Modifier le produit"
                  : "Ajouter un produit"}
              </h2>
            </div>

            <span className="rounded-full bg-pink-100 px-4 py-2 text-sm font-bold text-[#880E4F]">
              {editingId ? "Modification" : "Nouveau produit"}
            </span>

          </div>

          <form
            onSubmit={saveProduct}
            className="grid gap-5 md:grid-cols-2"
          >

            {/* NOM */}

            <div>
              <label className="mb-2 block text-sm font-semibold text-gray-700">
                Nom du produit
              </label>

              <input
                required
                placeholder="Ex : Savon naturel"
                value={form.name}
                onChange={(e) =>
                  setForm({
                    ...form,
                    name: e.target.value,
                  })
                }
                className="w-full rounded-xl border border-gray-200 p-4 outline-none transition-all duration-300 focus:border-[#C2185B] focus:ring-4 focus:ring-pink-100"
              />
            </div>

            {/* SLUG */}

            <div>
              <label className="mb-2 block text-sm font-semibold text-gray-700">
                Identifiant URL (Slug)
              </label>

              <input
                required
                placeholder="savon-naturel"
                value={form.slug}
                onChange={(e) =>
                  setForm({
                    ...form,
                    slug: e.target.value,
                  })
                }
                className="w-full rounded-xl border border-gray-200 p-4 outline-none transition-all duration-300 focus:border-[#C2185B] focus:ring-4 focus:ring-pink-100"
              />
            </div>

            {/* PRIX */}

            <div>
              <label className="mb-2 block text-sm font-semibold text-gray-700">
                Prix (FCFA)
              </label>

              <input
                required
                type="number"
                min="0"
                placeholder="1500"
                value={form.price}
                onChange={(e) =>
                  setForm({
                    ...form,
                    price: e.target.value,
                  })
                }
                className="w-full rounded-xl border border-gray-200 p-4 outline-none transition-all duration-300 focus:border-[#C2185B] focus:ring-4 focus:ring-pink-100"
              />
            </div>

            {/* STOCK */}

            <div>
              <label className="mb-2 block text-sm font-semibold text-gray-700">
                Quantité en stock
              </label>

              <input
                required
                type="number"
                min="0"
                placeholder="10"
                value={form.stock}
                onChange={(e) =>
                  setForm({
                    ...form,
                    stock: e.target.value,
                  })
                }
                className="w-full rounded-xl border border-gray-200 p-4 outline-none transition-all duration-300 focus:border-[#C2185B] focus:ring-4 focus:ring-pink-100"
              />
            </div>

            {/* CATÉGORIE */}

            <div>
              <label className="mb-2 block text-sm font-semibold text-gray-700">
                Catégorie
              </label>

              <select
                value={form.category_id}
                onChange={(e) =>
                  setForm({
                    ...form,
                    category_id: e.target.value,
                  })
                }
                className="w-full rounded-xl border border-gray-200 bg-white p-4 outline-none transition-all duration-300 focus:border-[#C2185B] focus:ring-4 focus:ring-pink-100"
              >
                <option value="">
                  Choisir une catégorie
                </option>

                {categories.map((category) => (
                  <option
                    key={category.id}
                    value={category.id}
                  >
                    {category.name}
                  </option>
                ))}
              </select>
            </div>

            {/* PHOTO */}

            <div className="rounded-2xl border-2 border-dashed border-pink-300 bg-pink-50/50 p-5 transition-all duration-300 hover:border-[#880E4F] hover:bg-pink-50">

              <label className="mb-3 block font-bold text-[#880E4F]">
                Photo du produit
              </label>

              <input
                type="file"
                accept="image/jpeg,image/png,image/webp,image/gif"
                onChange={uploadImage}
                disabled={uploadingImage}
                className="w-full cursor-pointer text-sm file:mr-3 file:rounded-lg file:border-0 file:bg-[#880E4F] file:px-4 file:py-2 file:font-semibold file:text-white hover:file:bg-[#C2185B]"
              />

              <p className="mt-3 text-xs text-gray-500">
                JPG, PNG, WEBP ou GIF · 5 Mo maximum
              </p>

              {uploadingImage && (
                <div className="mt-4 rounded-xl bg-white p-3">

                  <div className="mb-2 flex items-center gap-2 text-sm font-semibold text-[#880E4F]">
                    <span className="h-3 w-3 animate-spin rounded-full border-2 border-pink-200 border-t-[#880E4F]" />
                    Envoi de la photo...
                  </div>

                  <div className="h-2 overflow-hidden rounded-full bg-pink-100">
                    <div className="h-full w-1/2 animate-pulse rounded-full bg-[#C2185B]" />
                  </div>

                </div>
              )}

              {/* APERÇU PHOTO */}

              {imagePreview && (
                <div className="mt-5 animate-pulse rounded-2xl bg-white p-4 shadow-sm">

                  <p className="mb-3 text-sm font-bold text-[#880E4F]">
                    Aperçu de votre photo
                  </p>

                  <div className="flex flex-wrap items-center gap-4">

                    <img
                      src={imagePreview}
                      alt="Aperçu du produit"
                      className="h-36 w-36 rounded-2xl border-2 border-pink-100 object-cover shadow-lg transition-transform duration-500 hover:scale-105"
                    />

                    <div className="flex-1">

                      <p className="text-sm font-semibold text-green-700">
                        {form.image_url
                          ? "Photo enregistrée sur Supabase"
                          : "Aperçu local : envoi à effectuer"}
                      </p>

                      <button
                        type="button"
                        onClick={() => {
                          setForm((previous) => ({
                            ...previous,
                            image_url: "",
                          }));

                          setImagePreview("");
                          setMessage("Photo retirée.");
                        }}
                        className="mt-3 rounded-lg bg-red-50 px-4 py-2 text-sm font-semibold text-red-600 transition-all duration-300 hover:bg-red-100 hover:scale-105"
                      >
                        Retirer la photo
                      </button>

                    </div>
                  </div>
                </div>
              )}

            </div>

            {/* DESCRIPTION */}

            <div className="md:col-span-2">

              <label className="mb-2 block text-sm font-semibold text-gray-700">
                Description du produit
              </label>

              <textarea
                rows={4}
                placeholder="Présentez votre produit..."
                value={form.description}
                onChange={(e) =>
                  setForm({
                    ...form,
                    description: e.target.value,
                  })
                }
                className="w-full resize-y rounded-xl border border-gray-200 p-4 outline-none transition-all duration-300 focus:border-[#C2185B] focus:ring-4 focus:ring-pink-100"
              />

            </div>

            {/* VISIBILITÉ */}

            <label className="flex cursor-pointer items-center gap-3 rounded-xl bg-pink-50 p-4 transition-all duration-300 hover:bg-pink-100">

              <input
                type="checkbox"
                checked={form.active}
                onChange={(e) =>
                  setForm({
                    ...form,
                    active: e.target.checked,
                  })
                }
                className="h-5 w-5 accent-[#880E4F]"
              />

              <span className="font-semibold text-[#880E4F]">
                Produit visible dans la boutique
              </span>

            </label>

            {/* BOUTONS */}

            <div className="flex flex-wrap items-center gap-3 md:col-span-2">

              <button
                type="submit"
                disabled={loading || uploadingImage}
                className="rounded-xl bg-gradient-to-r from-[#880E4F] to-[#C2185B] px-7 py-4 font-bold text-white shadow-lg transition-all duration-300 hover:-translate-y-1 hover:scale-[1.02] hover:shadow-xl disabled:cursor-not-allowed disabled:opacity-50"
              >
                {loading
                  ? "Enregistrement..."
                  : editingId
                    ? "Enregistrer les modifications"
                    : "Ajouter le produit"}
              </button>

              {editingId && (
                <button
                  type="button"
                  onClick={cancelEditing}
                  className="rounded-xl border border-gray-300 px-6 py-4 font-semibold text-gray-600 transition-all duration-300 hover:bg-gray-100"
                >
                  Annuler
                </button>
              )}

            </div>

          </form>
        </section>

        {/* CATALOGUE */}

        <section className="rounded-3xl border border-pink-100 bg-white p-5 shadow-xl md:p-8">

          <div className="mb-7 flex flex-wrap items-center justify-between gap-4">

            <div>
              <p className="text-sm font-semibold uppercase tracking-wider text-[#C2185B]">
                Votre boutique
              </p>

              <h2 className="mt-2 text-2xl font-extrabold text-[#880E4F]">
                Catalogue des produits
              </h2>
            </div>

            <div className="rounded-2xl bg-gradient-to-r from-[#880E4F] to-[#C2185B] px-6 py-4 text-center text-white shadow-lg">
              <p className="text-3xl font-extrabold">
                {products.length}
              </p>

              <p className="text-xs font-semibold">
                Produits
              </p>
            </div>

          </div>

          {products.length === 0 && (
            <div className="rounded-2xl bg-pink-50 p-10 text-center">

              <p className="text-lg font-bold text-[#880E4F]">
                Aucun produit enregistré
              </p>

              <p className="mt-2 text-sm text-gray-500">
                Ajoutez votre premier produit avec le formulaire ci-dessus.
              </p>

            </div>
          )}

          <div className="grid gap-5 md:grid-cols-2">

            {products.map((product) => (
              <article
                key={product.id}
                className="group flex flex-col gap-4 overflow-hidden rounded-2xl border border-pink-100 bg-white p-4 shadow-sm transition-all duration-300 hover:-translate-y-1 hover:border-pink-300 hover:shadow-xl sm:flex-row"
              >

                {/* IMAGE */}

                <div className="relative h-40 w-full flex-shrink-0 overflow-hidden rounded-xl bg-pink-50 sm:h-32 sm:w-32">

                  {product.image_url ? (
                    <img
                      src={product.image_url}
                      alt={product.name}
                      className="h-full w-full object-cover transition-transform duration-500 group-hover:scale-110"
                    />
                  ) : (
                    <div className="flex h-full items-center justify-center text-sm text-gray-400">
                      Sans photo
                    </div>
                  )}

                </div>

                {/* INFORMATIONS */}

                <div className="min-w-0 flex-1">

                  <h3 className="text-lg font-bold text-[#880E4F]">
                    {product.name}
                  </h3>

                  <p className="mt-2 font-extrabold text-[#C2185B]">
                    {Number(product.price).toLocaleString("fr-FR")} FCFA
                  </p>

                  <p className="mt-1 text-sm text-gray-500">

function verifyPin(pin: string): boolean {
  const storedHash = process.env.ADMIN_PIN_HASH;

  if (!storedHash) {
    throw new Error("ADMIN_PIN_HASH manquant.");
  }

  const [algorithm, salt, hash] = storedHash.split("$");

  if (
    algorithm !== "scrypt" ||
    !salt ||
    !hash ||
    !/^[a-f0-9]{128}$/i.test(hash)
  ) {
    throw new Error("Format ADMIN_PIN_HASH invalide.");
  }

  const calculatedHash = scryptSync(pin, salt, 64);
  const expectedHash = Buffer.from(hash, "hex");

  return (
    calculatedHash.length === expectedHash.length &&
    timingSafeEqual(calculatedHash, expectedHash)
  );
}

  const [draft, setDraft] = useState("");

      writeStorage(STORAGE_KEY, id);
    }
    setConversationId(id);
    setCustomerName(readStorage(NAME_STORAGE_KEY) || "");
  }, []);

  const loadMessages = useCallback(async () => {
    if (!conversationId) return;
    try {
      const data = await requestJson<{ messages?: Message[] }>(

      setLoadError("");
    } catch (e) {
      setLoadError(errorMessage(e, "Connexion instable, nouvelle tentative…"));
    } finally {
      setLoadingMessages(false);
    }
  }, [conversationId]);

  useEffect(() => {
    if (!conversationId) return;
    void loadMessages();
    const timer = window.setInterval(() => void loadMessages(), MESSAGES_POLL_MS);
    return () => window.clearInterval(timer);
  }, [conversationId, loadMessages]);

  const updateCustomerName = useCallback((value: string) => {
    setCustomerName(value);
    writeStorage(NAME_STORAGE_KEY, value);

function MessageBubble({ message, variant }: { message: Message; variant: "client" | "admin" }) {
  const isUser = message.role === "user";
  const isAdminMsg = message.role === "admin";
  const isClientView = variant === "client";

  const tone = isUser
    ? `bg-[#2D6A4F] text-white ${isClientView ? "rounded-br-sm" : ""}`
    : isAdminMsg
      ? `border border-[#E7D9A8] bg-[#FFF9E8] text-[#493D1B] ${isClientView ? "rounded-bl-sm" : ""}`
      : `border border-[#D7EBDD] bg-white text-[#244333] ${isClientView ? "rounded-bl-sm" : ""}`;

  const label = isClientView
    ? isAdminMsg ? "Réponse de Gnilane" : "Conseiller beauté"
    : isUser ? "Cliente" : isAdminMsg ? "Gnilane" : "Conseiller beauté";

  const sizing = isClientView
    ? "max-w-[92%] rounded-2xl px-4 py-3 shadow-sm sm:max-w-[82%]"
    : "max-w-[90%] rounded-2xl px-4 py-3 text-sm leading-6";

  const products = isClientView ? message.metadata?.recommended_products : undefined;

  return (
    <div className={`flex ${isUser ? "justify-end" : "justify-start"}`}>
      <div className={`${sizing} ${tone}`}>
        {(!isClientView || !isUser) && (
          <p className="mb-1 text-[10px] font-bold uppercase tracking-wider opacity-65">{label}</p>
        )}
        <p className={`whitespace-pre-wrap ${isClientView ? "text-sm leading-7" : ""}`}>{message.content}</p>

        {products?.length ? (
          <div className="mt-3 grid gap-3 sm:grid-cols-2">
            {products.map((p) => (
              <article key={p.id} className="overflow-hidden rounded-xl border border-[#D7EBDD] bg-white text-[#1B4332]">
                <img src={p.image} alt={p.name} className="h-36 w-full object-cover" />
                <div className="p-3">
                  <p className="text-xs font-bold">{p.name}</p>
                  <p className="mt-1 text-sm font-bold text-[#9B7613]">{money(p.price)}</p>
                  <a href="/produits" className="mt-2 inline-flex rounded-full bg-[#2D6A4F] px-3 py-2 text-xs font-bold text-white">
                    Voir le produit
                  </a>
                </div>
              </article>
            ))}
          </div>
        ) : null}

        {isClientView && message.created_at && (
          <p className="mt-2 text-right text-[9px] opacity-55">
            {new Date(message.created_at).toLocaleString("fr-FR")}
          </p>
        )}
      </div>
    </div>
  );
}

/** Bulle de la cliente, du conseiller ou de Gnilane. */
function ChatBubble({ message }: { message: Message }) {
  const mine = message.role === "user";
  const fromGnilane = message.role === "admin";
  const products = message.metadata?.recommended_products ?? [];

  const bubbleTone = mine
    ? "rounded-br-md bg-[#2D6A4F] text-white"
    : fromGnilane
      ? "rounded-bl-md border border-[#E7D9A8] bg-[#FFFBEF] text-[#3F3414]"
      : "rounded-bl-md border border-[#D7EBDD] bg-white text-[#244333]";

  return (
    <div className={`cs-msg-in flex items-end gap-2 ${mine ? "justify-end" : "justify-start"}`}>
      {!mine &&
  (fromGnilane ? (
    <img
      src="/logo.jpg"
      alt=""
      aria-hidden="true"
      className="h-7 w-7 shrink-0 rounded-full border border-[#D4AF37] bg-white object-contain p-0.5"
    />
  ) : (
    <span
      className="grid h-7 w-7 shrink-0 place-items-center rounded-full bg-[#E5F2E8] text-[10px] font-bold text-[#2D6A4F]"
      aria-hidden="true"
    >
      ✿
    </span>
  ))}

      <div className={`flex max-w-[82%] flex-col sm:max-w-[70%] ${mine ? "items-end" : "items-start"}`}>
        {!mine && (
          <span className="mb-1 ml-1 text-[11px] font-semibold text-[#6B8B78]">
            {fromGnilane ? "Gnilane" : "Conseiller beauté"}
          </span>
        )}

        <div className={`whitespace-pre-wrap break-words rounded-2xl px-4 py-3 text-[15px] leading-7 shadow-sm ${bubbleTone}`}>
          {message.content}
        </div>

        {products.length > 0 && (
          <div className="mt-2 flex w-full flex-col gap-2">
            {products.map((p) => (
              <a
                key={p.id}
                href="/produits"
                className="flex items-center gap-3 rounded-2xl border border-[#D7EBDD] bg-white p-2 transition hover:-translate-y-0.5 hover:shadow-md"
              >
                <img src={p.image} alt={p.name} loading="lazy" className="h-14 w-14 shrink-0 rounded-xl object-cover" />
                <span className="min-w-0">
                  <span className="block truncate text-sm font-semibold">{p.name}</span>
                  <span className="text-xs font-bold text-[#94720F]">{money(p.price)}</span>
                </span>
              </a>
            ))}
          </div>
        )}

        {message.created_at && (
          <span className="mt-1 px-1 text-[10px] text-[#8A9D8E]">{formatTime(message.created_at)}</span>
        )}
      </div>
    </div>
  );
}

function ProductsSheet({
  open,
  onClose,
  onPick,
}: {
  open: boolean;
  onClose: () => void;
  onPick: (product: Product) => void;
}) {
  const [rendered, setRendered] = useState(open);

  useEffect(() => {
    if (open) {
      setRendered(true);
      return;
    }
    const timer = window.setTimeout(() => setRendered(false), 260);
    return () => window.clearTimeout(timer);
  }, [open]);

  useEffect(() => {
    if (!open) return;
    const onKey = (event: KeyboardEvent) => {
      if (event.key === "Escape") onClose();
    };
    const previousOverflow = document.body.style.overflow;
    document.body.style.overflow = "hidden";
    window.addEventListener("keydown", onKey);
    return () => {
      document.body.style.overflow = previousOverflow;
      window.removeEventListener("keydown", onKey);
    };
  }, [open, onClose]);

  if (!rendered) return null;

  return (
    <div className="fixed inset-0 z-50 flex items-end justify-center sm:items-center sm:p-6" role="dialog" aria-modal="true" aria-labelledby="cs-products-title">
      <button
        type="button"
        aria-label="Fermer la liste"
        onClick={onClose}
        className={`absolute inset-0 bg-[#0F2D20]/50 backdrop-blur-sm ${open ? "cs-backdrop-in" : "cs-backdrop-out"}`}
      />

      <div
        className={`relative flex max-h-[85dvh] w-full max-w-lg flex-col overflow-hidden rounded-t-[1.75rem] bg-white shadow-2xl sm:rounded-[1.75rem] ${
          open ? "cs-sheet-in" : "cs-sheet-out"
        }`}
        style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
      >
        <div className="mx-auto mt-2.5 h-1.5 w-12 shrink-0 rounded-full bg-[#CFE6D6] sm:hidden" />

        <div className="flex shrink-0 items-center justify-between px-5 pb-2 pt-3">
          <div>
            <h2 id="cs-products-title" className="font-serif text-xl font-bold">Nos produits</h2>
            <p className="mt-0.5 text-xs text-[#668071]">Choisissez un produit pour demander conseil à Gnilane.</p>
          </div>
          <button
            type="button"
            onClick={onClose}
            aria-label="Fermer"
            className="grid h-10 w-10 place-items-center rounded-full border border-[#CFE6D6] text-[#2D6A4F] transition hover:bg-[#EAF5ED]"
          >
            <X size={18} />
          </button>
        </div>

        <ul className="min-h-0 flex-1 space-y-2 overflow-y-auto px-4 pb-5 pt-2">
          {PRODUCTS.map((product, index) => (
            <li
              key={product.id}
              className="cs-item flex items-center gap-3 rounded-2xl border border-[#E3EEE6] bg-white p-2.5"
              style={{ animationDelay: `${Math.min(index, 8) * 40}ms` }}
            >
              <img src={product.image} alt={product.name} loading="lazy" className="h-14 w-14 shrink-0 rounded-xl object-cover" />
              <div className="min-w-0 flex-1">
                <p className="truncate text-sm font-semibold">{product.name}</p>
                <p className="text-xs font-bold text-[#94720F]">{money(product.price)}</p>
                {product.available_on_order && <p className="text-[10px] font-semibold text-[#85670E]">Sur commande</p>}
              </div>
              <button
                type="button"
                onClick={() => onPick(product)}
                className="shrink-0 rounded-full bg-[#2D6A4F] px-3.5 py-2 text-xs font-bold text-white transition hover:bg-[#1B4332]"
              >
                Demander
              </button>
            </li>
          ))}
        </ul>
      </div>
    </div>
  );
}

function ChatView({ chat, onLogoClick }: { chat: ChatState; onLogoClick: () => void }) {
  const [productsOpen, setProductsOpen] = useState(false);
  const [showName, setShowName] = useState(false);
  const bottomRef = useRef<HTMLDivElement>(null);
  const textareaRef = useRef<HTMLTextAreaElement>(null);

  const { draft, setDraft, messages, sending, loadingMessages, error, loadError, sendText } = chat;

  const closeProducts = useCallback(() => setProductsOpen(false), []);

  const pickProduct = useCallback(
    (product: Product) => {
      setDraft(`Bonjour Gnilane, je voudrais un conseil sur « ${product.name} ». `);
      setProductsOpen(false);
      window.setTimeout(() => textareaRef.current?.focus(), 320);
    },
    [setDraft],
  );

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
  }, [messages.length, sending]);

  useEffect(() => {
    const el = textareaRef.current;
    if (!el) return;
    el.style.height = "auto";
    el.style.height = `${Math.min(el.scrollHeight, 128)}px`;
  }, [draft]);

  const submit = (event?: FormEvent) => {
    event?.preventDefault();
    void sendText(draft);
  };

  const onKeyDown = (event: React.KeyboardEvent<HTMLTextAreaElement>) => {
    if (event.key === "Enter" && !event.shiftKey && !event.nativeEvent.isComposing) {
      event.preventDefault();
      submit();
    }
  };

  const canSend = !sending && draft.trim().length >= 2;
  const showSuggestions = !draft.trim() && !sending;
  const trimmedName = chat.customerName.trim();

  return (
    <main className="flex h-[100dvh] flex-col bg-[#F3F8F4] text-[#1B4332] sm:items-center sm:justify-center sm:p-4">
      <ChatStyles />

      <section className="flex h-full min-h-0 w-full max-w-3xl flex-col overflow-hidden bg-white sm:h-[min(820px,calc(100dvh_-_2rem))] sm:rounded-[2rem] sm:border sm:border-[#DCE9DD] sm:shadow-2xl">
        {/* En-tête */}
        <header className="flex shrink-0 items-center justify-between gap-3 border-b border-[#E3EEE6] px-4 py-3 sm:px-6 sm:py-4">
          <button type="button" onClick={onLogoClick} aria-label="Beauté Bio by Gnilane" className="flex min-w-0 items-center gap-3 text-left">
           <img
  src="/logo.jpg"
  alt="Logo Beauté Bio by Gnilane"
  className="h-11 w-11 shrink-0 rounded-full border-2 border-[#D4AF37] bg-white object-contain p-0.5 shadow-md"
/>
            <span className="min-w-0">
              <span className="block truncate font-serif text-base font-bold leading-tight">Beauté Bio by Gnilane</span>
              <span className="mt-0.5 flex items-center gap-1.5 text-[11px] font-semibold text-[#2D6A4F]">
                <span className="cs-online h-2 w-2 shrink-0 rounded-full bg-[#52B788]" />
                Conversation avec Gnilane
              </span>
            </span>
          </button>

          <div className="flex shrink-0 items-center gap-2">
            <button
              type="button"
              onClick={() => setProductsOpen(true)}
              className="inline-flex items-center gap-2 rounded-full bg-[#2D6A4F] px-4 py-2.5 text-xs font-bold text-white shadow-md transition hover:bg-[#1B4332]"
            >
              <ShoppingBag size={15} aria-hidden="true" />
              Produits
            </button>
            <a
              href="/"
              aria-label="Retour à la boutique"
              className="grid h-10 w-10 place-items-center rounded-full border border-[#CFE6D6] text-[#2D6A4F] transition hover:bg-[#EAF5ED]"
            >
              <ArrowLeft size={16} />
            </a>
          </div>
        </header>

        {/* Messages */}
        <div className="cs-scroll min-h-0 flex-1 space-y-4 overflow-y-auto overscroll-contain bg-[#FAFDFB] px-4 py-5 sm:px-6" aria-live="polite">
          {loadingMessages && messages.length === 0 && (
            <div className="flex justify-center py-10">
              <span className="cs-spin h-6 w-6 rounded-full border-2 border-[#2D6A4F] border-t-transparent" />
            </div>
          )}

          {!loadingMessages && messages.length === 0 && <Welcome />}

          {messages.map((message) => (
            <ChatBubble key={message.id} message={message} />
          ))}

          {sending && <TypingRow />}

          {loadError && <p className="text-center text-xs text-[#789082]">{loadError}</p>}

          <div ref={bottomRef} />
        </div>

        {/* Saisie */}
        <div
          className="shrink-0 border-t border-[#E3EEE6] bg-white px-4 pt-3 sm:px-6"
          style={{ paddingBottom: "max(0.875rem, env(safe-area-inset-bottom))" }}
        >
          {error && (
            <p role="alert" className="cs-shake mb-3 rounded-xl bg-red-50 px-3 py-2 text-xs font-semibold text-red-700">
              {error}
            </p>
          )}

          {showSuggestions && (
            <div className="cs-scroll mb-3 flex gap-2 overflow-x-auto pb-1 sm:flex-wrap" role="group" aria-label="Questions suggérées">
              {SUGGESTIONS.map((suggestion, index) => (
                <button
                  key={suggestion}
                  type="button"
                  onClick={() => void sendText(suggestion)}
                  className="cs-chip shrink-0 rounded-full border border-[#CFE6D6] bg-[#F4FAF5] px-3.5 py-2 text-left text-xs font-semibold leading-5 text-[#2D6A4F] transition hover:-translate-y-0.5 hover:border-[#2D6A4F] hover:bg-[#E5F2E8] active:scale-95"
                  style={{ animationDelay: `${index * 70}ms` }}
                >
                  {suggestion}
                </button>
              ))}
            </div>
          )}

          <form onSubmit={submit}>
            <div className="mb-2 flex items-center justify-between gap-2 text-[11px] text-[#6B8B78]">
              <button type="button" onClick={() => setShowName((v) => !v)} className="font-semibold text-[#2D6A4F] hover:underline">
                {trimmedName ? `Au nom de ${trimmedName}` : "Ajouter mon prénom"}
              </button>
              <span className={draft.length > 1300 ? "font-semibold text-[#B45309]" : ""}>
                {draft.length}/{MAX_MESSAGE_LENGTH}
              </span>
            </div>

            {showName && (
              <input
                autoFocus
                value={chat.customerName}
                onChange={(e) => chat.updateCustomerName(e.target.value)}
                placeholder="Votre prénom"
                aria-label="Votre prénom"
                maxLength={80}
                className="cs-pop mb-2 w-full rounded-xl border border-[#CFE6D6] px-3 py-2.5 text-sm outline-none focus:border-[#2D6A4F]"
              />
            )}

            <div className="flex items-end gap-2 rounded-2xl border border-[#CFE6D6] bg-[#FBFEFC] p-2 transition focus-within:border-[#2D6A4F] focus-within:ring-4 focus-within:ring-[#2D6A4F]/10">
              <textarea
                ref={textareaRef}
                value={draft}
                onChange={(e) => setDraft(e.target.value)}
                onKeyDown={onKeyDown}
                rows={1}
                maxLength={MAX_MESSAGE_LENGTH}
                placeholder="Écrivez votre message…"
                aria-label="Votre message"
                className="max-h-32 min-h-[40px] flex-1 resize-none bg-transparent px-2 py-2 text-[15px] leading-6 text-[#1B4332] outline-none placeholder:text-[#91A398]"
              />
              <button
                type="submit"
                disabled={!canSend}
                aria-label="Envoyer"
                className="cs-send grid h-10 w-10 shrink-0 place-items-center rounded-xl bg-[#2D6A4F] text-white shadow-md hover:bg-[#1B4332] disabled:cursor-not-allowed disabled:bg-[#B7CFC0] disabled:shadow-none"
              >
                {sending ? <span className="cs-spin h-4 w-4 rounded-full border-2 border-white border-t-transparent" /> : <Send size={17} />}
              </button>
            </div>

            <p className="mt-2 text-center text-[10px] text-[#8A9D8E]">
              Beaute bio by Gnilane à votre service.
            </p>
          </form>
        </div>
      </section>

      <ProductsSheet open={productsOpen} onClose={closeProducts} onPick={pickProduct} />
    </main>
  );
}

function AdminLoginModal({ admin }: { admin: ReturnType<typeof useAdmin> }) {
  const { closeLogin } = admin;

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => e.key === "Escape" && closeLogin();
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [closeLogin]);

  return (
    <div className="fixed inset-0 z-50 grid place-items-center bg-black/60 p-4" role="dialog" aria-modal="true" aria-labelledby="admin-title">
      <form onSubmit={admin.login} className="w-full max-w-sm rounded-3xl bg-white p-6 shadow-2xl">
        <div className="mb-4 flex items-center justify-between">
          <div>
            <p className="text-[10px] font-bold uppercase tracking-[.2em] text-[#2D6A4F]">Accès réservé</p>
            <h2 id="admin-title" className="mt-1 font-serif text-2xl font-bold">Espace Gnilane</h2>
          </div>
          <button type="button" onClick={closeLogin} className="beauty-flat rounded-full px-3 py-2 text-xl text-[#668071] hover:bg-[#F0F6F0]" aria-label="Fermer">
            ×
          </button>
        </div>
        <p className="mb-4 text-sm text-[#668071]">Saisissez votre code administrateur à {PIN_LENGTH} chiffres.</p>
        <input
          autoFocus
          type="password"
          inputMode="numeric"
          pattern={`[0-9]{${PIN_LENGTH}}`}
          maxLength={PIN_LENGTH}
          value={admin.pin}
          onChange={(e) => admin.setPin(e.target.value.replace(/\D/g, "").slice(0, PIN_LENGTH))}
          placeholder="••••••"
          aria-label="Code administrateur"
          className="w-full rounded-xl border border-[#CFE6D6] px-4 py-3 text-center text-2xl tracking-[.5em] outline-none focus:border-[#2D6A4F]"
        />
        {admin.error && <p role="alert" className="mt-3 text-xs text-red-600">{admin.error}</p>}
        <div className="mt-5 flex gap-2">
          <button type="button" onClick={closeLogin} className="flex-1 rounded-full border border-[#CFE6D6] px-4 py-3 text-sm font-bold text-[#2D6A4F]">
            Annuler
          </button>
          <button
            disabled={admin.loginLoading || admin.pin.length !== PIN_LENGTH}
            className="flex-1 rounded-full bg-[#2D6A4F] px-4 py-3 text-sm font-bold text-white disabled:opacity-50"
          >
            {admin.loginLoading ? "Vérification…" : "Se connecter"}
          </button>
        </div>
      </form>
    </div>
  );
}

function Reveal({
  children,
  delay = 0,
}: {
  children: ReactNode;
  delay?: number;
}) {
  const ref = useRef<HTMLDivElement>(null);
  const [visible, setVisible] = useState(false);

  useEffect(() => {
    const node = ref.current;
    if (!node) return;

    if (typeof IntersectionObserver === "undefined") {
      setVisible(true);
      return;
    }

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setVisible(true);
          observer.disconnect();
        }
      },
      { threshold: 0.1 }
    );

    observer.observe(node);
    return () => observer.disconnect();
  }, []);

  return (
    <div
      ref={ref}
      className={`bb-reveal ${visible ? "bb-in" : ""}`}
      style={{ transitionDelay: `${delay}ms` }}
    >
      {children}
    </div>
  );
}

function readCart(): CartItem[] {
  try {
    const saved = localStorage.getItem(CART_STORAGE_KEY);
    if (!saved) return [];

    const parsed: unknown = JSON.parse(saved);
    if (!Array.isArray(parsed)) return [];

    return parsed.filter((item): item is CartItem => {
      return (
        item !== null &&
        typeof item === "object" &&
        typeof item.id === "string" &&
        typeof item.name === "string" &&
        typeof item.price === "number" &&
        typeof item.quantity === "number" &&
        item.quantity > 0
      );
    });
  } catch {
    return [];
  }
}

// ==================================================
// GESTION DU PANIER
// ==================================================

function readCart(): CartItem[] {
  try {
    const saved = localStorage.getItem(CART_KEY);
    if (!saved) return [];

    const parsed: unknown = JSON.parse(saved);
    return Array.isArray(parsed) ? (parsed as CartItem[]) : [];
  } catch {
    return [];
  }
}

function countCart(cart: CartItem[]) {
  return cart.reduce(
    (sum, item) => sum + (Number(item.quantity) || 0),
    0
  );
}

function saveProductToCart(product: PreviewProduct): CartItem[] {
  const cart = readCart();
  const existing = cart.find((item) => item.id === product.id);

  let updatedCart: CartItem[];

  if (existing) {
    updatedCart = cart.map((item) =>
      item.id === product.id
        ? { ...item, quantity: item.quantity + 1 }
        : item
    );
  } else {
    updatedCart = [
      ...cart,
      {
        id: product.id,
        name: product.name,
        price: product.price,
        quantity: 1,
        image: product.image,
      },
    ];
  }

  localStorage.setItem(CART_KEY, JSON.stringify(updatedCart));
  return updatedCart;
}

function Reveal({
  children,
  className = "",
  delay = 0,
  as: Tag = "div",
}: {
  children: ReactNode;
  className?: string;
  delay?: number;
  as?: "div" | "article";
}) {
  const ref = useRef<HTMLDivElement & HTMLElement>(null);
  const [visible, setVisible] = useState(false);

  useEffect(() => {
    const node = ref.current;
    if (!node) return;

    if (typeof IntersectionObserver === "undefined") {
      setVisible(true);
      return;
    }

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setVisible(true);
          observer.disconnect();
        }
      },
      { threshold: 0.12, rootMargin: "0px 0px -6% 0px" }
    );

    observer.observe(node);
    return () => observer.disconnect();
  }, []);

  return (
    <Tag
      ref={ref as never}
      className={`bg-reveal ${visible ? "bg-reveal-in" : ""} ${className}`}
      style={{ transitionDelay: `${delay}s` }}
    >
      {children}
    </Tag>
  );
}

function getSecret() {
  const secret = process.env.ADMIN_SESSION_SECRET;
  if (!secret || secret.length < 32) {
    throw new Error("ADMIN_SESSION_SECRET doit contenir au moins 32 caractères.");
  }
  return secret;
}

function sign(value: string) {
  return createHmac("sha256", getSecret()).update(value).digest("hex");
}

export function createAdminToken() {
  const payload = Buffer.from(
    JSON.stringify({ role: "gnilane_admin", exp: Math.floor(Date.now() / 1000) + SESSION_SECONDS })
  ).toString("base64url");
  return `${payload}.${sign(payload)}`;
}

export function verifyAdminToken(token?: string | null) {
  if (!token) return false;
  const [payload, signature] = token.split(".");
  if (!payload || !signature) return false;
  const expected = sign(payload);
  const suppliedBuffer = Buffer.from(signature);
  const expectedBuffer = Buffer.from(expected);
  if (suppliedBuffer.length !== expectedBuffer.length) return false;
  if (!timingSafeEqual(suppliedBuffer, expectedBuffer)) return false;
  try {
    const data = JSON.parse(Buffer.from(payload, "base64url").toString("utf8"));
    return data.role === "gnilane_admin" && Number(data.exp) > Math.floor(Date.now() / 1000);
  } catch {
    return false;
  }
}

import { createContext, useContext, useEffect, useState } from "react";
import { Product, CartItem } from "@/types";

type CartContextType = {
  items: CartItem[];
  addItem: (
    product: Product,
    quantity?: number,
    variantName?: string,
    unitPrice?: number
  ) => void;
  updateQuantity: (productId: string, quantity: number) => void;
  removeItem: (productId: string) => void;
  clearCart: () => void;
  subtotal: number;
  itemCount: number;
};

const CartContext = createContext<CartContextType | null>(null);

const CART_KEY = "Beaute-bio-cart";

export function CartProvider({
  children,
}: {
  children: React.ReactNode;
}) {
  const [items, setItems] = useState<CartItem[]>([]);
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    try {
      const saved = localStorage.getItem(CART_KEY);

      if (saved) {
        const parsed = JSON.parse(saved);

        if (Array.isArray(parsed)) {
          setItems(parsed);
        }
      }
    } catch (error) {
      console.error("Erreur lors du chargement du panier :", error);
    } finally {
      setLoaded(true);
    }
  }, []);

  useEffect(() => {
    if (!loaded) return;

    try {
      localStorage.setItem(CART_KEY, JSON.stringify(items));
    } catch (error) {
      console.error("Erreur lors de la sauvegarde du panier :", error);
    }
  }, [items, loaded]);

  function addItem(
    product: Product,
    quantity: number = 1,
    variantName?: string,
    unitPrice?: number
  ) {
    const finalPrice =
      unitPrice ??
      product.price ??
      product.variants?.[0]?.price ??
      0;

    setItems((prev) => {
      const existing = prev.find(
        (item) =>
          item.product.id === product.id &&
          item.variantName === variantName
      );

      if (existing) {
        return prev.map((item) =>
          item.product.id === product.id &&
          item.variantName === variantName
            ? {
                ...item,
                quantity: item.quantity + quantity,
              }
            : item
        );
      }

      return [
        ...prev,
        {
          product,
          quantity,
          variantName,
          unitPrice: finalPrice,
        },
      ];
    });
  }

  function updateQuantity(
    productId: string,
    quantity: number
  ) {
    setItems((prev) =>
      quantity <= 0
        ? prev.filter(
            (item) => item.product.id !== productId
          )
        : prev.map((item) =>
            item.product.id === productId
              ? { ...item, quantity }
              : item
          )
    );
  }

  function removeItem(productId: string) {
    setItems((prev) =>
      prev.filter((item) => item.product.id !== productId)
    );
  }

  function clearCart() {
    setItems([]);
  }

  const subtotal = items.reduce(
    (total, item) =>
      total + item.unitPrice * item.quantity,
    0
  );

  const itemCount = items.reduce(
    (total, item) => total + item.quantity,
    0
  );

  return (
    <CartContext.Provider
      value={{
        items,
        addItem,
        updateQuantity,
        removeItem,
        clearCart,
        subtotal,
        itemCount,
      }}
    >
      {children}
    </CartContext.Provider>
  );
}

export function useCart() {
  const context = useContext(CartContext);

  if (!context) {
    throw new Error(
      "useCart doit être utilisé dans un CartProvider"
    );
  }

  return context;
}
